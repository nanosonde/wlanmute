cmake_minimum_required(VERSION 3.20)
project(wlanmute VERSION 0.3.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(wlanmute_policy src/network_policy.cpp)
target_include_directories(wlanmute_policy PUBLIC src)

function(wlanmute_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
wlanmute_warnings(wlanmute_policy)

if(WIN32)
  enable_language(RC)
  add_executable(wlanmute WIN32 src/main.cpp src/windows_network.cpp src/wlanmute.rc)
  target_compile_definitions(wlanmute PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  # Single source of truth for the version: shared by C++ and the resource compiler.
  configure_file(src/version.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/version.h" @ONLY)
  target_include_directories(wlanmute PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(wlanmute PRIVATE wlanmute_policy wlanapi iphlpapi ws2_32 ole32 oleaut32 uuid shell32 user32 comctl32)
  wlanmute_warnings(wlanmute)
  if(MSVC)
    # The manifest is embedded via wlanmute.rc; avoid a duplicate linker-generated one.
    target_link_options(wlanmute PRIVATE /MANIFEST:NO)
  elseif(MINGW)
    target_link_options(wlanmute PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(policy_tests tests/policy_tests.cpp)
  target_link_libraries(policy_tests PRIVATE wlanmute_policy)
  wlanmute_warnings(policy_tests)
  add_test(NAME network_policy COMMAND policy_tests)
endif()
